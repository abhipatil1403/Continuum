# 48. Program to Get the Maximum and Minimum Value in a Dictionary

my_dict={'a':'1', 'b':'2', 'c':'100', 'd':'4'}
max_value = max(int(value) for value in my_dict.values())
min_value = min(int(value) for value in my_dict.values())

print("Maximum value in dictionary:", max_value)
print("Minimum value in dictionary:", min_value)
