# 44. Program to Sort (Ascending and Descending) a Dictionary by Value

my_dict = {'a': 1, 'c': 3, 'b': 2,}
asc = dict(sorted(my_dict.items(), key=lambda item: item[1]))
desc = dict(sorted(my_dict.items(), key=lambda item: item[1], reverse=True))
print("Sorted (Ascending) Dictionary:", asc)
print("Sorted (Descending) Dictionary:", desc)
